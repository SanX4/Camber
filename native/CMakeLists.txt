cmake_minimum_required(VERSION 3.24)
project(CamberNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CAMBER_IMAGE_TESTS_ONLY "Build portable image tests without Windows SDK" OFF)
enable_testing()
add_executable(Camber.ImageTests tests/image_tests.cpp virtual-camera/frame_transform.cpp)
add_test(NAME image-transforms COMMAND Camber.ImageTests)
if(CAMBER_IMAGE_TESTS_ONLY)
  return()
endif()
if(NOT MSVC)
  message(FATAL_ERROR "Production targets require MSVC 2022 and Windows 11 SDK. Portable tests: -DCAMBER_IMAGE_TESTS_ONLY=ON")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set(WEBRTC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/libwebrtc/libwebrtc-x64-release" CACHE PATH "Pinned libwebrtc SDK directory")
if(NOT EXISTS "${WEBRTC_ROOT}/lib/libwebrtc.dll.lib")
  message(FATAL_ERROR "Run tools/bootstrap.ps1 to fetch the pinned libwebrtc SDK")
endif()
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(json)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_XAUDIO_WIN10 OFF CACHE BOOL "" FORCE)
FetchContent_Declare(directxtk GIT_REPOSITORY https://github.com/microsoft/DirectXTK.git GIT_TAG 5a8f5d01cc1328e6451b588617c9985fa5a2a8ab)
FetchContent_MakeAvailable(directxtk)
add_library(Camber.Camera SHARED virtual-camera/dllmain.cpp virtual-camera/media_source.cpp virtual-camera/frame_reader.cpp virtual-camera/frame_transform.cpp virtual-camera/Camber.Camera.def)
target_link_libraries(Camber.Camera PRIVATE mfplat mfuuid advapi32 runtimeobject)
add_executable(Camber.Media media/main.cpp media/reactions.cpp)
add_executable(Camber.ActivationTests tests/activation_tests.cpp)
target_link_libraries(Camber.ActivationTests PRIVATE mfplat mfuuid ole32 advapi32)
add_dependencies(Camber.ActivationTests Camber.Camera)
add_test(NAME camera-source-activation COMMAND Camber.ActivationTests)
# Manual integration check for machines with Camber installed. It opens only
# the Camber virtual source and is excluded from default CI/CTest runs.
add_executable(Camber.InstalledCameraTest tests/installed_camera_test.cpp)
target_link_libraries(Camber.InstalledCameraTest PRIVATE mf mfplat mfuuid mfreadwrite mfsensorgroup ole32 advapi32)
target_compile_definitions(Camber.Media PRIVATE LIB_WEBRTC_API_DLL RTC_DESKTOP_DEVICE)
target_include_directories(Camber.Media PRIVATE "${WEBRTC_ROOT}/include")
target_link_libraries(Camber.Media PRIVATE "${WEBRTC_ROOT}/lib/libwebrtc.dll.lib" nlohmann_json::nlohmann_json mfplat mfuuid mfsensorgroup ole32 advapi32 ws2_32 gdiplus)
add_custom_command(TARGET Camber.Media POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WEBRTC_ROOT}/lib/libwebrtc.dll" "$<TARGET_FILE_DIR:Camber.Media>")
add_test(NAME native-h264-codec COMMAND Camber.Media --probe)
add_executable(Camber.WebRtcLoopbackTests tests/webrtc_loopback_tests.cpp media/reactions.cpp)
target_compile_definitions(Camber.WebRtcLoopbackTests PRIVATE LIB_WEBRTC_API_DLL RTC_DESKTOP_DEVICE)
target_include_directories(Camber.WebRtcLoopbackTests PRIVATE "${WEBRTC_ROOT}/include")
target_link_libraries(Camber.WebRtcLoopbackTests PRIVATE "${WEBRTC_ROOT}/lib/libwebrtc.dll.lib" nlohmann_json::nlohmann_json mfplat mfuuid mfsensorgroup ole32 advapi32 ws2_32 gdiplus)
add_test(NAME generated-h264-receiver COMMAND Camber.WebRtcLoopbackTests)
set_tests_properties(generated-h264-receiver PROPERTIES TIMEOUT 30)
add_executable(Camber.ReactionTests tests/reaction_tests.cpp media/reactions.cpp)
target_link_libraries(Camber.ReactionTests PRIVATE gdiplus)
add_test(NAME reaction-composition COMMAND Camber.ReactionTests)

target_link_libraries(Camber.Media PRIVATE DirectXTK d3d11 dxgi)

target_link_libraries(Camber.WebRtcLoopbackTests PRIVATE DirectXTK d3d11 dxgi)

target_link_libraries(Camber.ReactionTests PRIVATE DirectXTK d3d11 dxgi)

add_executable(Camber.SecurityTests tests/security_tests.cpp)
target_link_libraries(Camber.SecurityTests PRIVATE advapi32)
add_test(NAME frame-object-acl COMMAND Camber.SecurityTests)

add_executable(Camber.UpgradeCheck installer/upgrade_check.cpp)
add_test(NAME installer-session-guard COMMAND Camber.UpgradeCheck --self-test)

target_link_libraries(Camber.Camera PRIVATE bcrypt)
target_link_libraries(Camber.Media PRIVATE bcrypt)

target_link_libraries(Camber.ActivationTests PRIVATE bcrypt)
add_executable(Camber.FrameProofTests tests/frame_proof_tests.cpp virtual-camera/frame_reader.cpp virtual-camera/frame_transform.cpp)
target_link_libraries(Camber.FrameProofTests PRIVATE mfplat mfuuid ole32 advapi32 bcrypt)
add_test(NAME frame-ownership-proof COMMAND Camber.FrameProofTests)

# Embed trusted shaders at build time; never load shader code from a user path.
find_program(CAMBER_FXC NAMES fxc.exe fxc HINTS "${DIRECTX_FXC_TOOL}")
if(NOT CAMBER_FXC AND DIRECTX_FXC_TOOL)
  set(CAMBER_FXC "${DIRECTX_FXC_TOOL}")
endif()
if(NOT CAMBER_FXC)
  message(FATAL_ERROR "Windows SDK FXC shader compiler is required")
endif()
set(ENHANCE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/enhance-generated")
file(MAKE_DIRECTORY "${ENHANCE_GENERATED}")
add_custom_command(OUTPUT "${ENHANCE_GENERATED}/enhance_vs.h" "${ENHANCE_GENERATED}/enhance_ps.h"
  COMMAND "${CAMBER_FXC}" /nologo /O3 /T vs_4_0 /E FullScreenVS /Vn EnhanceVS /Fh "${ENHANCE_GENERATED}/enhance_vs.h" "${CMAKE_CURRENT_SOURCE_DIR}/media/enhance.hlsl"
  COMMAND "${CAMBER_FXC}" /nologo /O3 /T ps_4_0 /E EnhancePS /Vn EnhancePS /Fh "${ENHANCE_GENERATED}/enhance_ps.h" "${CMAKE_CURRENT_SOURCE_DIR}/media/enhance.hlsl"
  DEPENDS media/enhance.hlsl VERBATIM)
add_library(Camber.Enhancement STATIC media/enhancement.cpp "${ENHANCE_GENERATED}/enhance_vs.h" "${ENHANCE_GENERATED}/enhance_ps.h")
target_include_directories(Camber.Enhancement PRIVATE "${ENHANCE_GENERATED}")
target_link_libraries(Camber.Enhancement PRIVATE d3d11 dxgi)
target_link_libraries(Camber.Media PRIVATE Camber.Enhancement)
target_link_libraries(Camber.WebRtcLoopbackTests PRIVATE Camber.Enhancement)
add_executable(Camber.EnhancementTests tests/enhancement_tests.cpp media/reactions.cpp)
target_link_libraries(Camber.EnhancementTests PRIVATE Camber.Enhancement DirectXTK d3d11 dxgi gdiplus)
add_test(NAME image-enhancement COMMAND Camber.EnhancementTests)
add_executable(Camber.Launcher WIN32 launcher/main.cpp launcher/launcher.rc)
target_link_libraries(Camber.Launcher PRIVATE user32)
target_link_options(Camber.Launcher PRIVATE /MANIFESTUAC:level='asInvoker')
# Entry points must work before the offline installer adds the C++ runtime.
set_property(TARGET Camber.Launcher Camber.UpgradeCheck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Directory.Build.props" CAMBER_VERSION_DEFINITION)
string(REGEX MATCH "<Version>([^<]+)</Version>" CAMBER_VERSION_MATCH "${CAMBER_VERSION_DEFINITION}")
set(CAMBER_BUILD_VERSION "${CMAKE_MATCH_1}")
string(REGEX MATCH "^([0-9]+)\.([0-9]+)\.([0-9]+)" CAMBER_VERSION_CORE "${CAMBER_BUILD_VERSION}")
set(CAMBER_VERSION_NUMERIC "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
foreach(CAMBER_BINARY_NAME IN ITEMS Camber.Launcher Camber.Media Camber.Camera Camber.UpgradeCheck)
  set(CAMBER_FILE_TYPE "VFT_APP")
  if(CAMBER_BINARY_NAME STREQUAL "Camber.Camera")
    set(CAMBER_FILE_TYPE "VFT_DLL")
  endif()
  configure_file(common/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${CAMBER_BINARY_NAME}.version.rc" @ONLY)
  target_sources(${CAMBER_BINARY_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${CAMBER_BINARY_NAME}.version.rc")
endforeach()
